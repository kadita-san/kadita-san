import os
import subprocess

def create_large_commit(filename="large_file.txt", num_additions=1000, commit_message="Add 1000 lines to demonstrate large commit."):
    """
    Creates a file with a specified number of lines, stages it for commit, and commits it to Git.

    Args:
        filename (str): The name of the file to create and commit. Defaults to "large_file.txt".
        num_additions (int): The number of lines to add to the file. Defaults to 1000.
        commit_message (str): The commit message. Defaults to "Add 1000 lines to demonstrate large commit.".
    """

    try:
        # Create a big text file
        with open(filename, "w") as f:
            for i in range(num_additions):
                f.write(f"This is line number {i+1} demonstrating a large addition to a file.\n")

        # Check if git repository exists. if no, initialize it
        if not os.path.exists(".git"):
            subprocess.run(["git", "init"])

        # Stage the file for commit
        subprocess.run(["git", "add", filename])

        # Commit the file
        subprocess.run(["git", "commit", "-m", commit_message])

        print(f"Successfully created and committed {filename} with {num_additions} additions.")

    except Exception as e:
        print(f"An error occurred: {e}")


if __name__ == "__main__":
    create_large_commit()
